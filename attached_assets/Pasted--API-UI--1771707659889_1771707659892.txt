### **فایل: `pages/api/test-provider.ts`** (برای Next.js)

```typescript
import type { NextApiRequest, NextApiResponse } from 'next';

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') {
    return res.status(405).json({ error: 'Method not allowed' });
  }

  const { provider, apiKey, prompt } = req.body;

  const configs: any = {
    openrouter: {
      url: 'https://openrouter.ai/api/v1/chat/completions',
      headers: {
        'Authorization': `Bearer ${apiKey}`,
        'Content-Type': 'application/json',
        'HTTP-Referer': process.env.NEXT_PUBLIC_SITE_URL || 'http://localhost:3000',
        'X-Title': 'Arman Law Firm'
      },
      body: {
        model: 'google/gemini-2.0-flash-exp:free',
        messages: [{ role: 'user', content: prompt }],
        max_tokens: 100
      }
    },
    poyo: {
      url: 'https://api.poyo.ai/v1/chat/completions',
      headers: {
        'Authorization': `Bearer ${apiKey}`,
        'Content-Type': 'application/json'
      },
      body: {
        model: 'gpt-4o-mini',
        messages: [{ role: 'user', content: prompt }],
        max_tokens: 100
      }
    },
    portkey: {
      url: 'https://api.portkey.ai/v1/chat/completions',
      headers: {
        'x-portkey-api-key': apiKey,
        'x-portkey-provider': 'openai',
        'Content-Type': 'application/json'
      },
      body: {
        model: 'gpt-4o-mini',
        messages: [{ role: 'user', content: prompt }],
        max_tokens: 100
      }
    }
  };

  const config = configs[provider];
  if (!config) {
    return res.status(400).json({ error: 'Invalid provider' });
  }

  const start = Date.now();

  try {
    const response = await fetch(config.url, {
      method: 'POST',
      headers: config.headers,
      body: JSON.stringify(config.body)
    });

    const data = await response.json();
    const duration = Date.now() - start;

    if (!response.ok) {
      throw new Error(data.error?.message || response.statusText);
    }

    res.json({
      success: true,
      duration,
      response: data.choices?.[0]?.message?.content || JSON.stringify(data)
    });
  } catch (error: any) {
    const duration = Date.now() - start;
    res.json({
      success: false,
      duration,
      error: error.message
    });
  }
}
```

```typescript
const testProvider = async (id: string) => {
  setTestingProvider(id);
  setTestResult(null);
  const start = Date.now();
  
  try {
    let apiKey = '';
    
    if (id === 'openrouter') apiKey = apiKeys.openrouter1;
    else if (id === 'poyo') apiKey = apiKeys.poyo1;
    else if (id === 'portkey') apiKey = apiKeys.portkey;

    // ✅ استفاده از API route به جای fetch مستقیم
    const res = await fetch('/api/test-provider', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        provider: id,
        apiKey,
        prompt: testPrompt
      })
    });

    const data = await res.json();
    const duration = Date.now() - start;

    if (data.success) {
      setTestResult({ 
        provider: id, 
        success: true, 
        duration: data.duration, 
        response: data.response 
      });
      addLog(id, 'test', 'success', data.duration);
    } else {
      throw new Error(data.error);
    }
  } catch (error: any) {
    const duration = Date.now() - start;
    setTestResult({ provider: id, success: false, error: error.message });
    addLog(id, 'test', 'error', duration, error.message);
  } finally {
    setTestingProvider(null);
  }
};
```

### **کامپوننت جدول خلاصه**

```typescript
const ResultsSummary = () => {
  const providerResults = providers.map(p => {
    const providerLogs = logs.filter(l => l.provider === p.id);
    const successCount = providerLogs.filter(l => l.status === 'success').length;
    const errorCount = providerLogs.filter(l => l.status === 'error').length;
    const avgLatency = successCount > 0
      ? Math.round(providerLogs.filter(l => l.status === 'success').reduce((acc, curr) => acc + curr.duration, 0) / successCount)
      : 0;

    return {
      name: p.name,
      status: errorCount === 0 && successCount > 0 ? '✅ کار می‌کند' : successCount > 0 ? '⚠️ جزئی' : '❌ خطا',
      tests: providerLogs.length,
      success: successCount,
      errors: errorCount,
      avgLatency: avgLatency > 0 ? `${avgLatency}ms` : '-'
    };
  });

  return (
    <div className="bg-gray-50 dark:bg-gray-700 rounded-lg p-4 mb-6">
      <h3 className="font-bold text-gray-900 dark:text-white mb-3">
        {isRtl ? '📊 خلاصه نتایج' : '📊 Results Summary'}
      </h3>
      <table className="w-full text-sm">
        <thead>
          <tr className="text-left border-b dark:border-gray-600">
            <th className="pb-2">{isRtl ? 'ارائه‌دهنده' : 'Provider'}</th>
            <th className="pb-2">{isRtl ? 'وضعیت' : 'Status'}</th>
            <th className="pb-2">{isRtl ? 'موفق/کل' : 'Success/Total'}</th>
            <th className="pb-2">{isRtl ? 'میانگین' : 'Avg Latency'}</th>
          </tr>
        </thead>
        <tbody>
          {providerResults.map(r => (
            <tr key={r.name} className="border-b dark:border-gray-600">
              <td className="py-2 font-medium">{r.name}</td>
              <td className="py-2">{r.status}</td>
              <td className="py-2">{r.success}/{r.tests}</td>
              <td className="py-2">{r.avgLatency}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
};

// استفاده در JSX:
{activeTab === 'providers' && (
  <>
    <ResultsSummary />
    {/* ... بقیه کد */}
  </>
)}
```
