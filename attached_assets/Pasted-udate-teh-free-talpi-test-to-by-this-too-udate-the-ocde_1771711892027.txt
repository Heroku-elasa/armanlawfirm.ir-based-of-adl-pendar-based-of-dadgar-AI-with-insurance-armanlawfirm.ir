# Helper to test models for OpenAI-compatible clients
# ────────────────────────────────────────────────
def test_models(client, models, api_name, key_short):
    results = {}
    for model in models:
        print(f"Testing {api_name} model '{model}' with key {key_short}...")
        try:
            response = client.chat.completions.create(
                model=model,
                messages=[{"role": "user", "content": f"Say 'Test OK for {model}'"}],
                max_tokens=20
            )
            results[model] = "WORKS"
            print(f"SUCCESS: {model} works! Response: {response.choices[0].message.content.strip()}")
        except Exception as e:
            results[model] = f"FAILED: {str(e)}"
            print(results[model])
    return results

# OpenRouter usage check (remaining credits & estimated reset time)
# ────────────────────────────────────────────────
def check_openrouter_usage(key, key_short):
    print(f"\n=== Checking OpenRouter usage for key {key_short} ===")
    headers = {"Authorization": f"Bearer {key}"}
    try:
        resp = requests.get("https://openrouter.ai/api/v1/key", headers=headers)
        resp.raise_for_status()
        data = resp.json()["data"]
        
        print(f"Remaining credits: {data.get('limit_remaining', 'Unlimited')}")
        print(f"Total usage (all time): {data['usage']}")
        print(f"Daily usage: {data['usage_daily']}")
        print(f"Is free tier: {data['is_free_tier']}")
        
        reset_type = data.get('limit_reset')
        if reset_type:
            now = datetime.now(timezone.utc)
            if reset_type == 'daily':
                next_reset = (now + timedelta(days=1)).replace(hour=0, minute=0, second=0, microsecond=0)
            elif reset_type == 'weekly':
                next_reset = now + timedelta(days=(7 - now.weekday()))  # Next Monday UTC
                next_reset = next_reset.replace(hour=0, minute=0, second=0, microsecond=0)
            elif reset_type == 'monthly':
                next_month = now.replace(day=28) + timedelta(days=4)  # Approx next month
                next_reset = next_month.replace(day=1, hour=0, minute=0, second=0, microsecond=0)
            else:
                next_reset = None
            
            if next_reset:
                time_remaining = next_reset - now
                print(f"Estimated time to reset ({reset_type}): {time_remaining}")
            else:
                print(f"Reset type: {reset_type} (no time estimate available)")
        else:
            print("No reset limit (unlimited or never resets)")
    except Exception as e:
        print(f"Usage check FAILED: {str(e)} (Key may be invalid)")

def test_poyo(key):
    key_short = key[:10] + "..."
    print(f"\n=== Testing Poyo key: {key_short} ===")
    headers = {
        "Authorization": f"Bearer {key}",
        "Content-Type": "application/json"
    }
    # 2026 models (image/video; fixed names like 'flux.2')
    models = ["flux.2", "nano-banana-pro", "seedream-4.5", "kling-3.0", "sora-2"]  # First 3 image, last 2 video
    results = {}
    print("Note: No API for usage remaining; check dashboard: https://poyo.ai/dashboard")
    
    for model in models:
        print(f"Testing Poyo model '{model}'...")
        payload = {
            "model": model,
            "input": {"prompt": "a simple test: red circle on white background"},  # Text-to-image/video test
            "callback_url": "https://example.com/webhook"
        }
        try:
            resp = requests.post(
                "https://api.poyo.ai/api/generate/submit",
                headers=headers,
                json=payload,
                timeout=15
            )
            resp.raise_for_status()
            data = resp.json()
            if data.get("code") == 200 and "task_id" in data.get("data", {}):
                task_id = data["data"]["task_id"]
                results[model] = f"WORKS (Task ID: {task_id})"
                print(results[model])
                time.sleep(2)
                status_resp = requests.get(f"https://api.poyo.ai/api/generate/status/{task_id}", headers=headers)
                if status_resp.status_code == 200:
                    print("Status:", status_resp.json())
                else:
                    print("Status failed:", status_resp.text)
            else:
                results[model] = f"FAILED: {data}"
                print(results[model])
        except Exception as e:
            results[model] = f"FAILED: {str(e)}"
            print(results[model])
    
    if all("FAILED" in v for v in results.values()):
        print("All failed - key invalid/expired? Get new: https://poyo.ai/dashboard/api-key")
    
    return results
