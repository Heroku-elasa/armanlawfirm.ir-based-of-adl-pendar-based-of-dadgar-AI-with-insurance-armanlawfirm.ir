## 🎯 اسکریپت خودکار (فقط API‌های جدید):

ذخیره کن به اسم `add_new_apis.sh`:

```bash
#!/bin/bash

echo "🔍 پیدا کردن API‌های جدید..."

# بیار تغییرات
git fetch origin main

# فایل‌های جدید API
NEW_API_FILES=$(git diff --name-only --diff-filter=A 8634aac origin/main | grep -E "(server|functions|api|routes)")

echo "📂 فایل‌های API جدید:"
echo "$NEW_API_FILES"

if [ -z "$NEW_API_FILES" ]; then
    echo "❌ هیچ API جدیدی پیدا نشد!"
    exit 0
fi

echo ""
read -p "می‌خوای اینارو اضافه کنی؟ (y/n) " -n 1 -r
echo

if [[ $REPLY =~ ^[Yy]$ ]]; then
    # بیار فایل‌های جدید
    for file in $NEW_API_FILES; do
        echo "➕ اضافه کردن: $file"
        git checkout origin/main -- "$file"
    done
    
    # کامیت
    git add .
    git commit -m "Added new API files from main"
    
    # تست
    echo "🧪 تست..."
    npm install
    npm run build
    
    if [ $? -eq 0 ]; then
        echo "✅ موفق! همه چی کار می‌کنه"
    else
        echo "❌ خطا! برمی‌گردیم..."
        git reset --hard HEAD~1
    fi
else
    echo "❌ لغو شد"
fi
```

اجرا:
```bash
chmod +x add_new_apis.sh
./add_new_apis.sh
```
