import requests

# ────────────────────────────────────────────────
# Helper to print table
# ────────────────────────────────────────────────
def print_table(headers, rows):
    col_widths = [max(len(str(row[i])) for row in rows + [headers]) for i in range(len(headers))]
    print("|" + "|".join(f" {h:<{w}} " for h, w in zip(headers, col_widths)) + "|")
    print("|" + "|".join("-" * (w + 2) for w in col_widths) + "|")
    for row in rows:
        print("|" + "|".join(f" {str(r):<{w}} " for r, w in zip(row, col_widths)) + "|")

# ────────────────────────────────────────────────
# Helper to test models for OpenAI-compatible
# ────────────────────────────────────────────────
def test_models(client, models, providers, api_name, key_short):
    results = []
    for model, provider in zip(models, providers):
        print(f"Testing {api_name} model '{model}' with key {key_short}...")
        try:
            extra_headers = {"x-portkey-provider": provider} if api_name == "Portkey" else {}
            response = client.chat.completions.create(
                model=model,
                messages=[{"role": "user", "content": f"Say 'Test OK for {model}'"}],
                max_tokens=20,
                extra_headers=extra_headers  # For Portkey
            )
            status = "WORKS"
            print(f"SUCCESS: {model} works! Response: {response.choices[0].message.content.strip()}")
        except Exception as e:
            status = f"FAILED: {str(e)}"
            print(status)
        results.append([model, status])
    return results

# ────────────────────────────────────────────────
# OpenRouter usage check
# ────────────────────────────────────────────────
def check_openrouter_usage(key, key_short):
    print(f"\n=== Checking OpenRouter usage for key {key_short} ===")
    headers = {"Authorization": f"Bearer {key}"}
    try:
        resp = requests.get("https://openrouter.ai/api/v1/key", headers=headers)
        resp.raise_for_status()
        data = resp.json()["data"]
        
        print(f"Remaining credits: {data.get('limit_remaining', 'Unlimited')}")
        print(f"Total usage (all time): {data['usage']}")
        print(f"Daily usage: {data['usage_daily']}")
        print(f"Is free tier: {data['is_free_tier']}")
        
        reset_type = data.get('limit_reset')
        if reset_type:
            now = datetime.now(timezone.utc)
            if reset_type == 'daily':
                next_reset = (now + timedelta(days=1)).replace(hour=0, minute=0, second=0, microsecond=0)
                time_rem = next_reset - now
                print(f"Estimated time to daily reset: {time_rem}")
            # Add weekly/monthly if needed
            else:
                print(f"Reset type: {reset_type} (manual estimate needed)")
        else:
            print("No reset limit (unlimited or never resets)")
        return True
    except Exception as e:
        print(f"Usage check FAILED: {str(e)}")
        return False

    for model in models:
        print(f"Testing Poyo model '{model}'...")
        payload = {
            "model": model,
            "input": {"prompt": "a simple test: red circle on white background"},
            "callback_url": "https://example.com/webhook"
        }
        try:
            resp = requests.post("https://api.poyo.ai/api/generate/submit", headers=headers, json=payload, timeout=15)
            resp.raise_for_status()
            data = resp.json()
            if data.get("code") == 200 and "task_id" in data.get("data", {}):
                task_id = data["data"]["task_id"]
                status = f"WORKS (Task ID: {task_id})"
                print(status)
                time.sleep(2)
                status_resp = requests.get(f"https://api.poyo.ai/api/generate/status/{task_id}", headers=headers)
                if status_resp.status_code == 200:
                    print("Status:", status_resp.json())
                else:
                    print("Status failed:", status_resp.text)
            else:
                status = f"FAILED: {data}"
                print(status)
        except Exception as e:
            status = f"FAILED: {str(e)}"
            print(status)
        results.append([model, status])
    
    if all("FAILED" in s[1] for s in results):
        print("All failed - key invalid/expired/no credits? Get new: https://poyo.ai/dashboard/api-key")
    
    print("\nPoyo Results Table:")

    return results
